cmake_minimum_required(VERSION 3.25)
project(linuxbee VERSION 0.8.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(LINUXBEE_BUILD_TESTS "Build the test programs (needs Qt6::Test)" OFF)
option(LINUXBEE_WERROR "Treat compiler warnings as errors" OFF)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets)
if(LINUXBEE_BUILD_TESTS)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
endif()

add_compile_options(-Wall -Wextra -Wpedantic)
if(LINUXBEE_WERROR)
  add_compile_options(-Werror)
endif()
add_compile_definitions(QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)

enable_testing()
add_subdirectory(src/core)
add_subdirectory(src/cli)
add_subdirectory(src/ui)
add_subdirectory(src/app)
if(LINUXBEE_BUILD_TESTS)
  add_subdirectory(tests)
endif()
add_subdirectory(packaging)
